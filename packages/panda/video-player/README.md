# @isbatak/panda-video-player

Panda CSS slot recipe and preset for the video player component from `@isbatak/zag-video-player`.

```ts
import { videoPlayerPreset } from "@isbatak/panda-video-player"
import { defineConfig } from "@pandacss/dev"

export default defineConfig({
  designSystem: "@isbatak/panda-ds",
  presets: [videoPlayerPreset],
})
```
