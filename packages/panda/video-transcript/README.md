# @isbatak/panda-video-transcript

Panda CSS slot recipe and preset for the video transcript component from `@isbatak/zag-video-transcript`.

```ts
import { videoTranscriptPreset } from "@isbatak/panda-video-transcript"
import { defineConfig } from "@pandacss/dev"

export default defineConfig({
  designSystem: "@isbatak/panda-ds",
  presets: [videoTranscriptPreset],
})
```
